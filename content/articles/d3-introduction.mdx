# Getting Started with D3.js

D3.js (Data-Driven Documents) is a powerful and flexible JavaScript library for creating data visualizations. Created by Mike Bostock, it's been the gold standard for web-based visualization since 2011.

<Callout type="info">
  **Interactive Examples Below!** Hover over elements to see tooltips and effects. Click **Run** to re-execute, **Fullscreen** to expand!
</Callout>

## What Makes D3 Special?

D3 isn't a charting library. It's a toolkit for **binding data to the DOM** and applying transformations:

- **Unlimited flexibility**: Create any visualization imaginable
- **Web standards**: Uses SVG, Canvas, and HTML directly
- **No black boxes**: You understand exactly what's happening
- **Smooth animations**: Built-in transition support
- **Interactivity**: Full access to DOM events

## 1. Bar Chart: Global Temperature Anomalies

Hover over bars to see details!

<D3 title="Global Temperature Anomaly" height={420} showCode={false}>
{`
const data = [
  { year: '1970', value: 0.03, color: '#3b82f6' },
  { year: '1980', value: 0.26, color: '#f97316' },
  { year: '1990', value: 0.45, color: '#f97316' },
  { year: '2000', value: 0.42, color: '#f97316' },
  { year: '2010', value: 0.72, color: '#ef4444' },
  { year: '2015', value: 0.90, color: '#ef4444' },
  { year: '2020', value: 1.02, color: '#dc2626' },
  { year: '2024', value: 1.29, color: '#b91c1c' },
];

const margin = { top: 40, right: 30, bottom: 60, left: 60 };
const innerWidth = width - margin.left - margin.right;
const innerHeight = height - margin.top - margin.bottom;

const svg = d3.select(container)
  .append('svg')
  .attr('width', width)
  .attr('height', height);

// Tooltip
const tooltip = d3.select(container)
  .append('div')
  .style('position', 'absolute')
  .style('background', 'rgba(0,0,0,0.9)')
  .style('border', '1px solid #64748b')
  .style('border-radius', '6px')
  .style('padding', '8px 12px')
  .style('color', 'white')
  .style('font-size', '13px')
  .style('font-weight', '500')
  .style('pointer-events', 'none')
  .style('opacity', 0)
  .style('transition', 'opacity 0.15s');

const chart = svg.append('g')
  .attr('transform', \`translate(\${margin.left},\${margin.top})\`);

// Scales
const x = d3.scaleBand()
  .domain(data.map(d => d.year))
  .range([0, innerWidth])
  .padding(0.2);

const y = d3.scaleLinear()
  .domain([0, d3.max(data, d => d.value)])
  .nice()
  .range([innerHeight, 0]);

// Gridlines
chart.append('g')
  .call(d3.axisLeft(y).ticks(5).tickSize(-innerWidth).tickFormat(() => ''))
  .selectAll('line').attr('stroke', '#334155').attr('stroke-opacity', 0.5);
chart.select('.domain').remove();

// Axes
chart.append('g')
  .attr('transform', \`translate(0,\${innerHeight})\`)
  .call(d3.axisBottom(x))
  .selectAll('text').attr('fill', 'white').attr('font-weight', 500);

chart.append('g')
  .call(d3.axisLeft(y).ticks(5))
  .selectAll('text').attr('fill', 'white').attr('font-weight', 500);

chart.selectAll('.domain, .tick line').attr('stroke', '#64748b');

// Bars with animation
const bars = chart.selectAll('rect')
  .data(data)
  .join('rect')
  .attr('x', d => x(d.year))
  .attr('width', x.bandwidth())
  .attr('y', innerHeight)
  .attr('height', 0)
  .attr('fill', d => d.color)
  .attr('rx', 4)
  .style('cursor', 'pointer');

bars.transition()
  .duration(800)
  .delay((d, i) => i * 100)
  .attr('y', d => y(d.value))
  .attr('height', d => innerHeight - y(d.value));

// Hover effects
bars.on('mouseover', function(event, d) {
  d3.select(this)
    .transition().duration(150)
    .attr('opacity', 0.8)
    .attr('transform', 'translateY(-3px)');
  
  tooltip
    .style('opacity', 1)
    .html(\`<strong>\${d.year}</strong><br/>+\${d.value.toFixed(2)}°C anomaly\`)
    .style('left', (event.offsetX + 10) + 'px')
    .style('top', (event.offsetY - 10) + 'px');
})
.on('mousemove', function(event) {
  tooltip
    .style('left', (event.offsetX + 10) + 'px')
    .style('top', (event.offsetY - 10) + 'px');
})
.on('mouseout', function() {
  d3.select(this)
    .transition().duration(150)
    .attr('opacity', 1)
    .attr('transform', 'translateY(0)');
  tooltip.style('opacity', 0);
});

// Title
svg.append('text')
  .attr('x', width / 2)
  .attr('y', 24)
  .attr('text-anchor', 'middle')
  .attr('fill', 'white')
  .attr('font-size', '16px')
  .attr('font-weight', '700')
  .text('Temperature Anomaly (°C vs 20th Century Avg)');

// Y-axis label

  .attr('transform', 'rotate(-90)')
  .attr('x', -height / 2)
  .attr('y', 16)
  .attr('text-anchor', 'middle')
  .attr('fill', 'white')
  .attr('font-size', '12px')
  .text('Anomaly (°C)');
`}
</D3>

## 2. Line Chart: Exoplanet Discoveries

Hover over data points to see values!

<D3 title="Exoplanet Discoveries" height={450} showCode={false}>
{`
const series = [
  { name: 'Transit', color: '#8b5cf6', data: [
    {x: 2005, y: 6}, {x: 2010, y: 130}, {x: 2012, y: 350}, {x: 2014, y: 1200},
    {x: 2016, y: 2700}, {x: 2018, y: 3100}, {x: 2020, y: 3400}, {x: 2024, y: 4200}
  ]},
  { name: 'Radial Velocity', color: '#06b6d4', data: [
    {x: 2005, y: 150}, {x: 2010, y: 460}, {x: 2012, y: 580}, {x: 2014, y: 650},
    {x: 2016, y: 720}, {x: 2018, y: 820}, {x: 2020, y: 920}, {x: 2024, y: 1080}
  ]},
  { name: 'Microlensing', color: '#f59e0b', data: [
    {x: 2005, y: 4}, {x: 2010, y: 12}, {x: 2012, y: 18}, {x: 2014, y: 35},
    {x: 2016, y: 52}, {x: 2018, y: 85}, {x: 2020, y: 145}, {x: 2024, y: 210}
  ]},
];

const margin = { top: 40, right: 120, bottom: 50, left: 60 };
const innerWidth = width - margin.left - margin.right;
const innerHeight = height - margin.top - margin.bottom;

const svg = d3.select(container)
  .append('svg')
  .attr('width', width)
  .attr('height', height);

// Tooltip
const tooltip = d3.select(container)
  .append('div')
  .style('position', 'absolute')
  .style('background', 'rgba(0,0,0,0.9)')
  .style('border', '1px solid #64748b')
  .style('border-radius', '6px')
  .style('padding', '8px 12px')
  .style('color', 'white')
  .style('font-size', '13px')
  .style('font-weight', '500')
  .style('pointer-events', 'none')
  .style('opacity', 0)
  .style('transition', 'opacity 0.15s');

const chart = svg.append('g')
  .attr('transform', \`translate(\${margin.left},\${margin.top})\`);

const allPoints = series.flatMap(s => s.data);

const x = d3.scaleLinear()
  .domain(d3.extent(allPoints, d => d.x))
  .range([0, innerWidth]);

const y = d3.scaleLinear()
  .domain([0, d3.max(allPoints, d => d.y)])
  .nice()
  .range([innerHeight, 0]);

// Gridlines
chart.append('g')
  .call(d3.axisLeft(y).ticks(5).tickSize(-innerWidth).tickFormat(() => ''))
  .selectAll('line').attr('stroke', '#334155').attr('stroke-opacity', 0.5);
chart.select('.domain').remove();

// Axes
chart.append('g')
  .attr('transform', \`translate(0,\${innerHeight})\`)
  .call(d3.axisBottom(x).tickFormat(d3.format('d')))
  .selectAll('text').attr('fill', 'white').attr('font-weight', 500);

chart.append('g')
  .call(d3.axisLeft(y))
  .selectAll('text').attr('fill', 'white').attr('font-weight', 500);

chart.selectAll('.domain, .tick line').attr('stroke', '#64748b');

// Line generator
const line = d3.line()
  .x(d => x(d.x))
  .y(d => y(d.y))
  .curve(d3.curveMonotoneX);

// Area generator
const area = d3.area()
  .x(d => x(d.x))
  .y0(innerHeight)
  .y1(d => y(d.y))
  .curve(d3.curveMonotoneX);

// Draw series
series.forEach(s => {
  // Area fill
  chart.append('path')
    .datum(s.data)
    .attr('fill', s.color)
    .attr('fill-opacity', 0.1)
    .attr('d', area);

  // Line with animation
  const path = chart.append('path')
    .datum(s.data)
    .attr('fill', 'none')
    .attr('stroke', s.color)
    .attr('stroke-width', 2.5)
    .attr('d', line);

  const totalLength = path.node().getTotalLength();
  path.attr('stroke-dasharray', \`\${totalLength} \${totalLength}\`)
    .attr('stroke-dashoffset', totalLength)
    .transition().duration(1500).ease(d3.easeLinear)
    .attr('stroke-dashoffset', 0);

  // Dots with hover
  const dots = chart.selectAll(\`.dot-\${s.name.replace(/\\s/g, '')}\`)
    .data(s.data)
    .join('circle')
    .attr('class', \`dot-\${s.name.replace(/\\s/g, '')}\`)
    .attr('cx', d => x(d.x))
    .attr('cy', d => y(d.y))
    .attr('r', 0)
    .attr('fill', s.color)
    .attr('stroke', 'white')
    .attr('stroke-width', 2)
    .style('cursor', 'pointer');

  dots.transition().delay(1500).duration(300).attr('r', 5);

  dots.on('mouseover', function(event, d) {
    d3.select(this)
      .transition().duration(150)
      .attr('r', 10);
    
    tooltip
      .style('opacity', 1)
      .html(\`<strong>\${s.name}</strong><br/>\${d.x}: \${d.y.toLocaleString()} planets\`)
      .style('left', (event.offsetX + 15) + 'px')
      .style('top', (event.offsetY - 10) + 'px');
  })
  .on('mouseout', function() {
    d3.select(this).transition().duration(150).attr('r', 5);
    tooltip.style('opacity', 0);
  });
});

// Legend
const legend = svg.append('g')
  .attr('transform', \`translate(\${width - margin.right + 10}, \${margin.top})\`);

series.forEach((s, i) => {
  const g = legend.append('g').attr('transform', \`translate(0, \${i * 25})\`);
  g.append('line').attr('x1', 0).attr('x2', 20).attr('stroke', s.color).attr('stroke-width', 2.5);
  g.append('text').attr('x', 25).attr('y', 4).attr('fill', 'white').attr('font-size', '12px').attr('font-weight', 500).text(s.name);
});

// Title
svg.append('text')
  .attr('x', width / 2).attr('y', 24)
  .attr('text-anchor', 'middle')
  .attr('fill', 'white').attr('font-size', '16px').attr('font-weight', '700')
  .text('Cumulative Exoplanet Discoveries');
`}
</D3>

// Gear path generator
function gear(teeth, radius, annulus = false) {
  const n = teeth;
  const toothR = 0.008;
  let r2 = Math.abs(radius), r0 = r2 - toothR, r1 = r2 + toothR, r3 = 0.02;
  if (annulus) { r3 = r0; r0 = r1; r1 = r3; r3 = r2 + toothR * 3; }
  const da = Math.PI / n;
  let a = -Math.PI / 2 + (annulus ? da : 0);
  const p = ['M', r0 * Math.cos(a), r0 * Math.sin(a)];
  for (let i = 0; i < n; i++) {
    p.push('A', r0, r0, '0 0 1', r0 * Math.cos(a += da), r0 * Math.sin(a));
    p.push('L', r2 * Math.cos(a), r2 * Math.sin(a));
    p.push('L', r1 * Math.cos(a += da/3), r1 * Math.sin(a));
    p.push('A', r1, r1, '0 0 1', r1 * Math.cos(a += da/3), r1 * Math.sin(a));
    p.push('L', r2 * Math.cos(a += da/3), r2 * Math.sin(a));
    p.push('L', r0 * Math.cos(a), r0 * Math.sin(a));
  }
  p.push('M0', -r3, 'A', r3, r3, '0 0 0 0', r3, 'A', r3, r3, '0 0 0 0', -r3, 'Z');
  return p.join(' ');
}

## The D3 Mental Model

Understanding D3 requires thinking about **data binding**:

```javascript
d3.select('svg')
  .selectAll('circle')
  .data(stars)
  .join('circle')
  .attr('cx', d => xScale(d.mass))
  .attr('cy', d => yScale(d.luminosity))
  .attr('r', d => sizeScale(d.radius))
```

The `join()` method handles the **enter-update-exit** pattern:
- **Enter** creates new elements for new data points
- **Update** modifies existing elements when data changes
- **Exit** removes elements for data points that no longer exist
