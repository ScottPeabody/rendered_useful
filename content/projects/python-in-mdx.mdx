import numpy as np
import matplotlib.pyplot as plt

def mandelbrot(h, w, max_iter=100):
    y, x = np.ogrid[-1.4:1.4:h*1j, -2:0.8:w*1j]
    c = x + y*1j
    z = c
    divtime = max_iter + np.zeros(z.shape, dtype=int)
    
    for i in range(max_iter):
        z = z**2 + c
        diverge = z*np.conj(z) > 2**2
        div_now = diverge & (divtime == max_iter)
        divtime[div_now] = i
        z[diverge] = 2
    
    return divtime

plt.figure(figsize=(10, 8))

### Lorenz Attractor

The famous "butterfly effect" - a chaotic system that never repeats but stays bounded:

<Python title="Lorenz Attractor" code={`
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

def lorenz(xyz, s=10, r=28, b=2.667):
    x, y, z = xyz
    return np.array([s*(y - x), x*(r - z) - y, x*y - b*z])

dt = 0.01
num_steps = 10000

xs, ys, zs = np.empty(num_steps), np.empty(num_steps), np.empty(num_steps)
xs[0], ys[0], zs[0] = 0.1, 0, 0

for i in range(1, num_steps):
    d = lorenz([xs[i-1], ys[i-1], zs[i-1]]) * dt
    xs[i] = xs[i-1] + d[0]
    ys[i] = ys[i-1] + d[1]
    zs[i] = zs[i-1] + d[2]

fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')
ax.plot(xs, ys, zs, lw=0.5, color='#8b5cf6')
ax.set_xlabel('X')
ax.set_ylabel('Y')
ax.set_zlabel('Z')
ax.set_title('Lorenz Attractor')
plt.tight_layout()
plt.show()
`} />

### Conway's Game of Life

Cellular automata - simple rules create complex emergent behavior:

<Python title="Game of Life" code={`
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.colors import ListedColormap

def life_step(grid):
    neighbors = (
        np.roll(np.roll(grid, 1, 0), 1, 1) + np.roll(np.roll(grid, 1, 0), -1, 1) +
        np.roll(np.roll(grid, -1, 0), 1, 1) + np.roll(np.roll(grid, -1, 0), -1, 1) +
        np.roll(grid, 1, 0) + np.roll(grid, -1, 0) +
        np.roll(grid, 1, 1) + np.roll(grid, -1, 1)
    )
    return ((grid == 1) & ((neighbors == 2) | (neighbors == 3))) | ((grid == 0) & (neighbors == 3))

np.random.seed(42)
size = 50
grid = np.random.choice([0, 1], size=(size, size), p=[0.85, 0.15])

frames = [grid.copy()]
for _ in range(100):
    grid = life_step(grid).astype(int)
    frames.append(grid.copy())

fig, axes = plt.subplots(2, 4, figsize=(12, 6))
cmap = ListedColormap(['#1a1a2e', '#22d3ee'])
steps = [0, 10, 25, 50, 75, 100, 90, 80]

for ax, step in zip(axes.flat, steps):
    ax.imshow(frames[step], cmap=cmap)
    ax.set_title(f'Step {step}')
    ax.axis('off')

plt.suptitle("Conway's Game of Life", fontsize=14)
plt.tight_layout()
plt.show()

import numpy as np
import matplotlib.pyplot as plt

def draw_tree(ax, x, y, angle, length, depth, max_depth):
    if depth > max_depth or length < 1:
        return
    
    x2 = x + length * np.cos(np.radians(angle))
    y2 = y + length * np.sin(np.radians(angle))
    
    t = depth / max_depth
    color = (0.4 * (1-t) + 0.2 * t, 0.25 * (1-t) + 0.7 * t, 0.1 * (1-t) + 0.3 * t)
    linewidth = max(1, (max_depth - depth) * 0.8)
    
    ax.plot([x, x2], [y, y2], color=color, linewidth=linewidth)
    
    spread = 25 + np.random.uniform(-5, 5)
    shrink = 0.7 + np.random.uniform(-0.05, 0.05)
    
    draw_tree(ax, x2, y2, angle + spread, length * shrink, depth + 1, max_depth)
    draw_tree(ax, x2, y2, angle - spread, length * shrink, depth + 1, max_depth)

np.random.seed(123)
fig, ax = plt.subplots(figsize=(10, 10))
ax.set_facecolor('#1a1a2e')
fig.patch.set_facecolor('#1a1a2e')

draw_tree(ax, 0, 0, 90, 60, 0, 10)

ax.set_aspect('equal')

### Julia Set

A companion to the Mandelbrot. Each point in the Mandelbrot defines a unique Julia set:

<Python title="Julia Set" code={`
import numpy as np
import matplotlib.pyplot as plt

def julia(h, w, c, max_iter=200):
    y, x = np.ogrid[-1.5:1.5:h*1j, -1.5:1.5:w*1j]
    z = x + y*1j
    divtime = max_iter + np.zeros(z.shape, dtype=int)
    
    for i in range(max_iter):
        z = z**2 + c
        diverge = np.abs(z) > 2
        div_now = diverge & (divtime == max_iter)
        divtime[div_now] = i
        z[diverge] = 2
    
    return divtime

fig, axes = plt.subplots(2, 2, figsize=(10, 10))
c_values = [
    (-0.7 + 0.27015j, 'c = -0.7 + 0.27015i'),
    (-0.8 + 0.156j, 'c = -0.8 + 0.156i'),
    (-0.4 + 0.6j, 'c = -0.4 + 0.6i'),
    (0.285 + 0.01j, 'c = 0.285 + 0.01i'),
]

for ax, (c, title) in zip(axes.flat, c_values):
    img = julia(400, 400, c)
    ax.imshow(img, cmap='twilight_shifted', extent=[-1.5, 1.5, -1.5, 1.5])
    ax.set_title(title, fontsize=10)
    ax.axis('off')

plt.suptitle('Julia Sets', fontsize=14)
plt.tight_layout()
plt.show()
`} />

### Fourier Circle Drawing

<Python title="Fourier Epicycles" code={`
import numpy as np
import matplotlib.pyplot as plt

t = np.linspace(0, 2*np.pi, 1000)
n_terms = 15

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

y_total = np.zeros_like(t)
colors = plt.cm.viridis(np.linspace(0, 1, n_terms))

    y_term = (4/np.pi) * np.sin(n * t) / n

        ax1.plot(t, y_total, color=colors[i], alpha=0.5, label=f'{i+1} terms')

import numpy as np
from bokeh.plotting import figure, show
from bokeh.models import ColumnDataSource
from bokeh.palettes import Viridis256

np.random.seed(123)

# Define network architecture
layers = [4, 8, 6, 4, 2]  # neurons per layer
layer_names = ['Input', 'Hidden 1', 'Hidden 2', 'Hidden 3', 'Output']

# Generate node positions
nodes_x, nodes_y, nodes_layer, nodes_idx, nodes_activation = [], [], [], [], []
for i, n_neurons in enumerate(layers):
    x = i * 2
    y_positions = np.linspace(-(n_neurons-1)/2, (n_neurons-1)/2, n_neurons)
    for j, y in enumerate(y_positions):
        nodes_x.append(x)
        nodes_y.append(y)
        nodes_layer.append(layer_names[i])
        nodes_idx.append(j)
        nodes_activation.append(np.random.uniform(0.1, 1.0))

edges_x0, edges_y0, edges_x1, edges_y1, edges_weight = [], [], [], [], []
node_idx = 0
for i in range(len(layers) - 1):
    n_current = layers[i]
    n_next = layers[i + 1]

## How It Works

The `<Python>` component uses [Pyodide](https://pyodide.org/) to run Python in WebAssembly:

1. **First run** loads Pyodide (~10MB) from CDN
2. **Shared instance** - all Python blocks on a page share the same runtime
3. **Matplotlib integration** - figures captured as PNG and displayed inline
4. **Package support** - NumPy, Pandas, SciPy, Bokeh, and more
